package com.eignex.kumulant.bench

import com.eignex.koblas.DenseMatrix
import com.eignex.kumulant.math.factorCholesky
import kotlin.math.abs

private fun oldReconstruction(matrix: DenseMatrix, lower: DenseMatrix): Double {
    var error = 0.0
    var scale = 0.0
    for (j in 0 until matrix.cols) {
        for (i in j until matrix.rows) {
            var actual = 0.0
            for (k in 0..j) actual += lower[i, k] * lower[j, k]
            error = maxOf(error, abs(actual - matrix[i, j]))
            scale = maxOf(scale, abs(matrix[i, j]))
        }
    }
    return error / scale
}

private fun oldBackward(matrix: DenseMatrix, right: DenseMatrix, solution: DenseMatrix): Double {
    var matrixNorm = 0.0
    for (i in 0 until matrix.rows) {
        var rowSum = 0.0
        for (j in 0 until matrix.cols) rowSum += abs(matrix[i, j])
        matrixNorm = maxOf(matrixNorm, rowSum)
    }
    var error = 0.0
    for (column in 0 until right.cols) {
        var residual = 0.0
        var solutionNorm = 0.0
        var rightNorm = 0.0
        for (i in 0 until matrix.rows) {
            var actual = 0.0
            for (j in 0 until matrix.cols) actual += matrix[i, j] * solution[j, column]
            residual = maxOf(residual, abs(actual - right[i, column]))
            solutionNorm = maxOf(solutionNorm, abs(solution[i, column]))
            rightNorm = maxOf(rightNorm, abs(right[i, column]))
        }
        error = maxOf(error, residual / (matrixNorm * solutionNorm + rightNorm))
    }
    return error
}

fun main() {
    var cases = 0
    for (kind in listOf("precision", "rbf", "matern")) for (size in listOf(8, 63, 64, 65, 127, 128, 129, 512)) for (scale in listOf(1e-100, 1.0, 1e100)) {
        val matrix = CholeskyFixtures.matrix("$kind:$size")
        for (i in matrix.values.indices) matrix.values[i] *= scale
        val factor = matrix.factorCholesky()
        val right = CholeskyFixtures.right(size, 16)
        val solution = factor.solve(right)
        val lower = factor.lowerFactor()
        check(oldReconstruction(matrix, lower).toRawBits() == relativeReconstructionError(matrix, lower).toRawBits()) { "$kind $size $scale recon" }
        check(oldBackward(matrix, right, solution).toRawBits() == solveBackwardError(matrix, right, solution).toRawBits()) { "$kind $size $scale backward" }
        cases++
    }
    println("bitwise identical oracle results: $cases cases")
}
